### 一、标签处理与数据增强核心概念解析
#### 1. 微调数据标签处理
标签（即SFT数据中的`output`字段）是模型学习的核心目标，处理的核心是**准确性**和**一致性**：
- **准确性**：输出答案必须与问题强匹配，无事实错误（如问“图书馆开放时间”，输出不能是食堂的时间）；
- **一致性**：输出格式/表述风格统一（如时间统一为“X:XX-X:XX”、单位统一为“层/元/小时”、语气统一为客观陈述）。

#### 2. 简单数据增强（针对大模型微调）
数据增强的目的是扩充数据集规模、提升模型泛化能力，核心原则是**保持语义不变**（问题语义一致、输出答案不变），常用方法：
- **同义词替换**：替换问题中的非核心词汇（如“开放时间”→“营业时段”、“申请条件”→“申报要求”），不改变问题核心语义；
- **句式变换**：调整问题语序（如“食堂有几层？”→“食堂的楼层数量是多少？”）、主动/被动转换（适合部分场景）、反问改陈述，保持问题核心诉求不变。

### 二、实操：校园问答数据增强（扩充至1000条）
#### 前置依赖安装
需要用到`synonyms`（中文同义词替换）和`jieba`（分词），执行以下命令安装：


#### 完整代码实现
```python
import json
import random
import synonyms
import jieba
from typing import List, Dict, Set

# ---------------------- 1. 标签处理函数 ----------------------
def clean_output_label(output: str) -> str:
    """
    标签（output）处理：保证准确性、一致性
    - 去除多余空格/换行
    - 统一时间格式（如“8点-22点”→“8:00-22:00”）
    - 统一单位表述（如“3层楼”→“3层”）
    - 去除无意义语气词（如“哦、呢、啊”）
    """
    if not isinstance(output, str):
        return ""
    
    # 基础清洗：去空格、换行、语气词
    output = output.strip().replace("\n", "").replace("\r", "")
    tone_words = ["哦", "呢", "啊", "吧", "啦", "嘛"]
    for word in tone_words:
        output = output.replace(word, "")
    
    # 统一时间格式（简单匹配，可根据实际场景扩展）
    time_mapping = {
        "8点": "8:00", "9点": "9:00", "10点": "10:00",
        "11点": "11:00", "12点": "12:00", "13点": "13:00",
        "14点": "14:00", "15点": "15:00", "16点": "16:00",
        "17点": "17:00", "18点": "18:00", "19点": "19:00",
        "20点": "20:00", "21点": "21:00", "22点": "22:00",
        "23点": "23:00", "0点": "0:00", "凌晨0点": "0:00"
    }
    for old, new in time_mapping.items():
        output = output.replace(old, new)
    
    # 统一单位表述
    unit_mapping = {
        "层楼": "层", "个楼层": "层", "元钱": "元",
        "小时长": "小时", "天时间": "天"
    }
    for old, new in unit_mapping.items():
        output = output.replace(old, new)
    
    return output

# ---------------------- 2. 数据增强函数 ----------------------
def synonym_replacement(text: str, replace_ratio: float = 0.2) -> str:
    """
    同义词替换：替换非核心词汇，保持语义不变
    Args:
        text: 原始问题文本
        replace_ratio: 替换比例（默认替换20%的词汇）
    Returns:
        同义词替换后的文本
    """
    if not text:
        return text
    
    # 分词
    words = jieba.lcut(text)
    if len(words) <= 1:  # 单字/单词不替换
        return text
    
    # 筛选可替换的非核心词（排除标点、疑问词）
    stop_words = {"？", "：", "，", "。", "吗", "呢", "什么", "哪", "几", "多少", "何时", "哪里"}
    replaceable_words = [i for i, word in enumerate(words) if word not in stop_words]
    # 计算需要替换的词数
    replace_num = max(1, int(len(replaceable_words) * replace_ratio))
    # 随机选择要替换的词
    selected_indices = random.sample(replaceable_words, min(replace_num, len(replaceable_words)))
    
    # 同义词替换
    new_words = words.copy()
    for idx in selected_indices:
        word = new_words[idx]
        # 获取同义词（synonyms.nearby返回(同义词列表, 相似度)）
        syns = synonyms.nearby(word)[0]
        if syns and len(syns) > 0:
            # 随机选一个同义词（排除自身）
            syn = random.choice([s for s in syns if s != word])
            new_words[idx] = syn
    
    return "".join(new_words)

def sentence_paraphrase(text: str) -> str:
    """
    句式变换：调整语序/改写句式，保持语义不变
    示例：
    - “图书馆开放时间是多少？” → “图书馆的开放时间是多少？”
    - “食堂有几层？” → “食堂的楼层数量是多少？”
    - “奖学金怎么申请？” → “如何申请奖学金？”
    """
    if not text:
        return text
    
    # 定义句式变换规则（可根据校园问答场景扩展）
    paraphrase_rules = [
        # 原模式 → 新模式
        (r"^(.*)是多少\？$", r"\1的数量是多少？"),
        (r"^(.*)有几层\？$", r"\1的楼层数量是多少？"),
        (r"^(.*)怎么申请\？$", r"如何申请\1？"),
        (r"^(.*)开放时间\？$", r"\1的开放时间是多少？"),
        (r"^(.*)在哪里\？$", r"\1的位置在哪里？"),
    ]
    
    import re
    for pattern, repl in paraphrase_rules:
        if re.match(pattern, text):
            return re.sub(pattern, repl, text)
    
    # 无匹配规则时，微调语序（仅针对短文本）
    if len(text) > 5 and text.endswith("？"):
        words = jieba.lcut(text[:-1])  # 去掉问号分词
        if len(words) >= 3:
            # 交换前两个词（如“图书馆 开放时间”→“开放时间 图书馆”）
            words[0], words[1] = words[1], words[0]
            return "".join(words) + "？"
    
    return text

def augment_single_sample(sample: Dict[str, str]) -> List[Dict[str, str]]:
    """
    单样本增强：生成多个变体（同义词替换+句式变换）
    Args:
        sample: 原始样本 {"question": "...", "answer": "..."}
    Returns:
        增强后的样本列表（包含原始样本+变体）
    """
    augmented_samples = []
    original_question = sample["question"]
    original_answer = clean_output_label(sample["answer"])  # 先清洗标签
    
    # 生成原始样本（清洗标签后）
    augmented_samples.append({
        "question": original_question,
        "answer": original_answer
    })
    
    # 生成同义词替换变体
    syn_question = synonym_replacement(original_question)
    augmented_samples.append({
        "question": syn_question,
        "answer": original_answer
    })
    
    # 生成句式变换变体
    para_question = sentence_paraphrase(original_question)
    augmented_samples.append({
        "question": para_question,
        "answer": original_answer
    })
    
    # 生成“同义词+句式变换”组合变体
    syn_para_question = sentence_paraphrase(syn_question)
    augmented_samples.append({
        "question": syn_para_question,
        "answer": original_answer
    })
    
    return augmented_samples

# ---------------------- 3. 批量增强到指定数量 ----------------------
def batch_augment_data(
    raw_data: List[Dict[str, str]],
    target_num: int = 1000
) -> List[Dict[str, str]]:
    """
    批量增强数据到指定数量
    Args:
        raw_data: 原始数据列表
        target_num: 目标数据量（默认1000）
    Returns:
        增强后的去重数据列表
    """
    augmented_data = []
    # 去重集合（避免重复样本）
    seen_questions = set()
    
    # 循环增强直到达到目标数量
    while len(augmented_data) < target_num:
        for sample in raw_data:
            if len(augmented_data) >= target_num:
                break
            # 单样本增强
            variants = augment_single_sample(sample)
            for var in variants:
                if len(augmented_data) >= target_num:
                    break
                # 去重：问题未出现过才加入
                if var["question"] not in seen_questions:
                    seen_questions.add(var["question"])
                    augmented_data.append(var)
    
    # 截断到目标数量（防止超出）
    augmented_data = augmented_data[:target_num]
    
    # 转换为SFT格式（instruction/input/output）
    sft_data = []
    for item in augmented_data:
        sft_data.append({
            "instruction": f"请回答以下问题：{item['question']}",
            "input": "",
            "output": item["answer"]
        })
    
    return sft_data

# ---------------------- 4. 主函数：数据增强流程 ----------------------
if __name__ == "__main__":

    print(f"数据增强完成！共生成 {len(augmented_sft_data)} 条样本")

    for i in range(5):

        print(f"instruction: {augmented_sft_data[i]['instruction']}")

    # 统计标签一致性（示例）
    print("\n=== 标签处理效果示例 ===")
    test_answer = "食堂有3个楼层哦！"

    print(f"清洗后标签：{cleaned_answer}")
```

### 三、代码关键部分解释
#### 1. 标签处理（`clean_output_label`）
- **核心逻辑**：先清洗无意义字符（空格、语气词），再统一格式（时间、单位），确保输出答案的准确性和一致性；
- **扩展建议**：可根据实际场景添加更多格式规则（如金额、地址格式统一）。

#### 2. 数据增强核心函数
- **同义词替换（`synonym_replacement`）**：
  - 用`jieba`分词后，筛选非核心词（排除标点、疑问词），按比例替换为同义词（基于`synonyms`库）；
  - 避免替换核心词（如“图书馆”“奖学金”），防止语义改变。
- **句式变换（`sentence_paraphrase`）**：
  - 基于正则匹配改写句式（如“怎么申请”→“如何申请”）；
  - 短文本微调语序，保持语义不变。

#### 3. 批量增强（`batch_augment_data`）
- 循环增强单样本，直到达到1000条目标；
- 去重处理：避免重复问题样本；
- 转换为SFT微调格式（`instruction/input/output`），直接适配微调流程。

### 四、使用前置条件
1. 环境：Python 3.7+；
2. 依赖安装：执行`pip install synonyms jieba`；
3. 原始数据：需为`[{"question": "...", "answer": "..."}]`格式的列表，可根据实际场景扩展基础数据量。

### 总结
1. **标签处理核心**：
   - 准确性：保证输出答案与问题匹配，无事实错误；
   - 一致性：统一输出格式（时间、单位）和表述风格，去除无意义字符。
2. **数据增强关键**：
   - 同义词替换、句式变换需保持问题语义不变，输出答案完全一致；
   - 增强后需去重，避免重复样本降低训练效率。
3. **实操要点**：
   - 基础数据无需过多（示例中10条），通过增强可扩充至1000条；
   - 增强后的数据直接转换为SFT格式，可直接用于大模型微调。
